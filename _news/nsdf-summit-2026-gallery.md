  let photos = [],
    visible = [],
    current = 0,
    filter = "All photos",
    returnFocus;
  const dialog = document.querySelector(".nsdf-gallery-embed dialog"),
    grid = document.querySelector(".nsdf-gallery-embed #grid");
  function render() {
    visible = photos.filter(
      (p) => filter === "All photos" || p.category === filter,
    );
    document.querySelector(".nsdf-gallery-embed #count").textContent =
      visible.length + " photographs";
    grid.replaceChildren();
    visible.forEach((p, i) => {
      const b = document.createElement("button");
      b.className = "card";
      b.setAttribute("aria-label", "View photo " + p.id + " · " + p.category);
      const im = new Image();
      im.src = p.thumb;
      im.alt = p.category + " · Summit photograph " + p.id;
      im.loading = "lazy";
      im.width = 720;
      im.height = 180;
      const s = document.createElement("span");
      s.textContent = p.category + " · " + String(p.id).padStart(3, "0");
      b.append(im, s);
      b.onclick = () => {
        returnFocus = b;
        current = i;
        show();
        dialog.showModal();
      };
      grid.append(b);
    });
  }
  function show() {
    const p = visible[current];
    const large = document.querySelector(".nsdf-gallery-embed #large");
    large.src = p.src;
    large.alt = p.category + " · Summit photograph " + p.id;
    document.querySelector(".nsdf-gallery-embed #position").textContent =
      current + 1 + " / " + visible.length;
    document.querySelector(".nsdf-gallery-embed #caption").textContent = p.name;
    const dl = document.querySelector(".nsdf-gallery-embed #download");
    dl.href = p.src;
    dl.download = p.name;
  }
  function move(n) {
    current = (current + n + visible.length) % visible.length;
    show();
  }
  document.querySelector(".nsdf-gallery-embed #prev").onclick = () => move(-1);
  document.querySelector(".nsdf-gallery-embed #next").onclick = () => move(1);
  document.querySelector(".nsdf-gallery-embed #close").onclick = () => dialog.close();
  dialog.addEventListener("close", () => returnFocus?.focus());
  dialog.addEventListener("keydown", (e) => {
    if (e.key === "ArrowLeft") {
      e.preventDefault();
      move(-1);
    }
    if (e.key === "ArrowRight") {
      e.preventDefault();
      move(1);
    }
  });
  dialog.addEventListener("click", (e) => {
    const rect = dialog.getBoundingClientRect();
    if (e.target === dialog && e.clientY < rect.top) dialog.close();
  });
  photos = JSON.parse(
    document.querySelector("#photo-data").textContent,
  );
  [
    "All photos",
    "Talks & sessions",
    "Community & discussion",
    "Gathering & reception",

    const b = document.createElement("button");
    b.textContent = label;
    b.className = i ? "" : "active";
    b.setAttribute("aria-pressed", !i);
    b.onclick = () => {
      filter = label;
      document.querySelectorAll(".nsdf-gallery-embed .filters button").forEach((x) => {
        x.classList.toggle("active", x === b);
        x.setAttribute("aria-pressed", x === b);
      });
      render();
    };
    document.querySelector(".nsdf-gallery-embed .filters").append(b);
  });
